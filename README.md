# မြန်မာဝဘ်ဆိုဒ် ဖန်တီးခြင်း
မောင်း "myanmar_web"

# ခေါင်းစဉ်စာမျက်နှာ
ပြုလုပ် မြန်မာဝဘ်ဆိုဒ်(ခေါင်းစဉ်="မြန်မာရုပ်ရှင်ကမ္ဘာ")

    # ၁။ ခေါင်းပိုင်း
    ဌာနခွဲ ခေါင်းပိုင်း(အတန်း="ဆိုက်ခေါင်းစဉ်"):
        h1("🎬 မြန်မာရုပ်ရှင် စုစည်းမှု")
        p("ရှေးဟောင်းမှ ခေတ်သစ်ထိ မြန်မာရုပ်ရှင်များ")
        
        # ရှာဖွေရေးနေရာ
        ပုံစံ input(
            အမျိုးအစား="ရှာဖွေရန်",
            placeholder="ရုပ်ရှင်အမည် ရှာပါ...",
            id="ရှာဖွေရန်"
        )
    
    # ၂။ အဓိကအကြောင်းအရာ
    ဌာနခွဲ အကြောင်းအရာပိုင်း:
        h2("ကျော်ကြားရုပ်ရှင်များ")
        
        # ရုပ်ရှင်ကတ်များ
        အတွက် ရုပ်ရှင် ထံမှ ရုပ်ရှင်စာရင်း:
            ပြုလုပ် ရုပ်ရှင်ကတ်(ရုပ်ရှင်အချက်အလက်=ရုပ်ရှင်):
                img(
                    အရံ=ရုပ်ရှင်["ကြော်ငြာ"],
                    alt=ရုပ်ရှင်["ခေါင်းစဉ်"],
                    အနံ=၂၅၀
                )
                ဌာနခွဲ ကတ်အကြောင်းအရာ:
                    h3(ရုပ်ရှင်["ခေါင်းစဉ်"])
                    p(f"ဒါရိုက်တာ: {ရုပ်ရှင်['ဒါရိုက်တာ']}")
                    p(f"ခုနှစ်: {ရုပ်ရှင်['ခုနှစ်']}")
                    p(ရုပ်ရှင်["အမှတ်သင်္ကေတ"])
                    button(
                        "အသေးစိတ်ကြည့်ရန်",
                        onclick=f"အသေးစိတ်ကြည့်ရန်('{ရုပ်ရှင်ခေါင်းစဉ်}')"
                    )
